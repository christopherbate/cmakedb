cmake_minimum_required(VERSION 3.20)
project(scopes NONE)
set(TOP_VAL parent)
function(write_out out)
  set(inner local)
  set(${out} "from-function" PARENT_SCOPE)
endfunction()
write_out(RESULT)
message(STATUS "result=${RESULT}")
macro(mac_set)
  set(from_macro 1)
endmacro()
mac_set()
message(STATUS "leak=${from_macro}")
block()
  set(TOP_VAL shadowed)
  message(STATUS "in-block=${TOP_VAL}")
endblock()
message(STATUS "after-block=${TOP_VAL}")
add_subdirectory(sub)

  add_library(extra INTERFACE)
  add_library(base INTERFACE)
  target_link_libraries(base INTERFACE extra)
endif()
