cmake_minimum_required(VERSION 3.20)
project(whynot C)
option(ENABLE_COMPRESSION "gate the zlib_stub link" OFF)
add_library(app STATIC app.c)
add_library(zlib_stub STATIC z.c)
add_library(other STATIC z.c)
target_link_libraries(other PRIVATE zlib_stub)
if(ENABLE_COMPRESSION)
  target_link_libraries(app PRIVATE zlib_stub)
endif()
function(add_wrapped_tool name)
  add_executable(${name} app.c)
endfunction()
if(ENABLE_COMPRESSION)
  add_wrapped_tool(ztool)
endif()
function(define_extra_target)
  add_library(extra STATIC z.c)
endfunction()
set(_unused_gate OFF)
if(_unused_gate)
  set(SPECIAL_FLAG "on")
endif()
