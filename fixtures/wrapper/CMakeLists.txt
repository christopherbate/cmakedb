cmake_minimum_required(VERSION 3.20)
project(wrapper C)
function(add_project_library name)
  cmake_parse_arguments(APL "" "" "SOURCES;DEPS" ${ARGN})
  add_library(${name} STATIC ${APL_SOURCES})
  if(APL_DEPS)
    target_link_libraries(${name} PUBLIC ${APL_DEPS})
  endif()
endfunction()
add_project_library(core SOURCES core.c)
add_project_library(engine SOURCES engine.c DEPS core)
